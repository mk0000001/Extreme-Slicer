cmake_minimum_required(VERSION 3.20)
project(OrcaCudaOpenGLControl LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "This diagnostic requires Windows MSVC x64")
endif()
if(CMAKE_CXX_COMPILER_VERSION VERSION_LESS 19.44 OR NOT CMAKE_CXX_COMPILER_VERSION VERSION_LESS 19.45)
    message(FATAL_ERROR "This control requires the app's MSVC 19.44 toolchain")
endif()
set(BOOST_HEADER_DIR "" CACHE PATH "Verified Boost 1.84.0 source root")
if(NOT EXISTS "${BOOST_HEADER_DIR}/boost/algorithm/string/split.hpp")
    message(FATAL_ERROR "BOOST_HEADER_DIR must contain the verified Boost headers")
endif()
add_executable(orca-cuda-opengl-control WIN32 OrcaSlicer_cuda_opengl_control.cpp)
target_compile_features(orca-cuda-opengl-control PRIVATE cxx_std_17)
target_compile_definitions(orca-cuda-opengl-control PRIVATE SLIC3R_GUI=1 SLIC3R_WRAPPER_NOCONSOLE=1 UNICODE _UNICODE _CRT_SECURE_NO_WARNINGS)
target_include_directories(orca-cuda-opengl-control PRIVATE "${BOOST_HEADER_DIR}")
target_compile_options(orca-cuda-opengl-control PRIVATE /W4 /Zi /EHsc)
target_link_options(orca-cuda-opengl-control PRIVATE /DEBUG /OPT:REF /OPT:ICF)
target_link_libraries(orca-cuda-opengl-control PRIVATE opengl32 shell32 user32 gdi32)
